import Testing

@Suite("Load entities")
struct LoadEntitiesTests {
    @Test("Returns repository values")
    func returnsValues() async throws {
        let expected = [Entity(name: "Example")]

        #expect(result == expected)
    }
}

private struct StubRepository: EntityRepository {
    let values: [Entity]
    func entities() async throws -> [Entity] { values }
}
